## Executive Summary
This review covers the test suite for the AdaptiveCard web component, including UI tests, main component tests, and test setup utilities. The tests follow a TDD (Test-Driven Development) approach with a Red-Green-Refactor cycle, as evidenced by the comments. The overall structure is solid, but there are several critical and important issues that need to be addressed for production-ready testing.

### 2. Race Condition in Async Tests
**Location**: `adaptive-card.test.ts` lines 39-44, 311-318

**Issue**: `waitForCustomElement` implementation may not properly wait for component initialization.

**Current Code**:
```typescript
// setup.ts line 127-135
export const waitForCustomElement = async (element: HTMLElement): Promise<void> => {
  if ('connectedCallback' in element && typeof element.connectedCallback === 'function') {
    await new Promise(resolve => {
      requestAnimationFrame(() => {
        requestAnimationFrame(resolve);
      });
    });
  }
};
```

**Recommended Fix**:
```typescript
export const waitForCustomElement = async (
  element: HTMLElement, 
  timeout: number = 1000
): Promise<void> => {
  const startTime = Date.now();
  
  while (!element.shadowRoot && Date.now() - startTime < timeout) {
    await new Promise(resolve => requestAnimationFrame(resolve));
  }
  
  if (!element.shadowRoot) {
    throw new Error(`Custom element did not initialize within ${timeout}ms`);
  }
  
  // Wait for component lifecycle
  await customElements.whenDefined(element.tagName.toLowerCase());
  
  // Allow for async initialization
  await new Promise(resolve => setTimeout(resolve, 0));
};
```

### 3. Memory Leak in Test Setup
**Location**: `setup.ts` lines 191-199

**Issue**: Global event listeners are added but never removed, potentially causing memory leaks in test runs.

**Current Code**:
```typescript
beforeEach(() => {
  vi.spyOn(console, 'error').mockImplementation(() => {});
  vi.spyOn(console, 'warn').mockImplementation(() => {});
});

afterEach(() => {
  vi.restoreAllMocks();
  document.body.innerHTML = '';
});
```

**Recommended Fix**:
```typescript
let errorHandler: ((event: ErrorEvent) => void) | null = null;
let unhandledRejectionHandler: ((event: PromiseRejectionEvent) => void) | null = null;

beforeEach(() => {
  vi.spyOn(console, 'error').mockImplementation(() => {});
  vi.spyOn(console, 'warn').mockImplementation(() => {});
  
  errorHandler = (event) => {
    console.error('Test error:', event.error);
  };
  unhandledRejectionHandler = (event) => {
    console.error('Test promise rejection:', event.reason);
  };
  
  globalThis.addEventListener('error', errorHandler);
  globalThis.addEventListener('unhandledrejection', unhandledRejectionHandler);
});

afterEach(() => {
  vi.restoreAllMocks();
  document.body.innerHTML = '';
  
  if (errorHandler) {
    globalThis.removeEventListener('error', errorHandler);
  }
  if (unhandledRejectionHandler) {
    globalThis.removeEventListener('unhandledrejection', unhandledRejectionHandler);
  }
  
  // Clean up any remaining custom elements
  document.querySelectorAll('adaptive-card').forEach(el => el.remove());
});
```

### 4. Incomplete Mock Implementations
**Location**: `setup.ts` lines 8-44

**Issue**: Mock implementations don't properly simulate observer behavior.

**Current Implementation**:
```typescript
const ResizeObserverMock = vi.fn(() => ({
  observe: vi.fn(),
  unobserve: vi.fn(),
  disconnect: vi.fn(),
}));
```

**Improved Implementation**:
```typescript
class ResizeObserverMock {
  private callback: ResizeObserverCallback;
  private observedElements = new Set<Element>();
  
  constructor(callback: ResizeObserverCallback) {
    this.callback = callback;
  }
  
  observe(element: Element): void {
    this.observedElements.add(element);
    // Trigger initial observation
    setTimeout(() => {
      const entries: ResizeObserverEntry[] = [{
        target: element,
        contentRect: element.getBoundingClientRect(),
        borderBoxSize: [{ blockSize: 0, inlineSize: 0 }],
        contentBoxSize: [{ blockSize: 0, inlineSize: 0 }],
        devicePixelContentBoxSize: [{ blockSize: 0, inlineSize: 0 }]
      }];
      this.callback(entries, this);
    }, 0);
  }
  
  unobserve(element: Element): void {
    this.observedElements.delete(element);
  }
  
  disconnect(): void {
    this.observedElements.clear();
  }
}

vi.stubGlobal('ResizeObserver', ResizeObserverMock);
```

import { test } from '@percy/vitest';

test('AdaptiveCard visual states', async () => {
  const variants = ['elevated', 'outlined', 'filled'];
  const states = ['default', 'hover', 'focus', 'disabled', 'selected'];

    for (const state of states) {
      const card = document.createElement('adaptive-card');
      card.setAttribute('variant', variant);
